Function 关注和一般(新数据数组,原表数组,线路,计划结束月份,电压等级) 
    工作内容保存 = ""
    计划编号保存 = ""
    实际开始时间 = ""
    实际结束时间 = ""
    线路名 = 原表数组[线路][0]
    sRet = Regex.Replace(线路名, "乙", ".*?乙", 1)
    计划结束日期 = 原表数组[线路][2]
    工作内容合并 = ""
    For i = 0 To UBound(新数据数组) step 1 //i为最新数据中符合当前项目筛选条件的内容的下标
        工作地点和内容 = 新数据数组[i][1]+新数据数组[i][2]
        // 正则匹配线路名和日期
        bret = Regex.Test(工作地点和内容,sRet)
        实际时间 = Time.CDate(新数据数组[i][4], "yyyy.mm.dd")
        实际月份 = Time.Month(实际时间)
        If 实际月份 = 需要匹配的月份
            If 电压等级 = "110kV" //一年两测
                If bret=True And 计划结束月份<=6 And 实际月份<=6
                    回填 = 完成状态回填(新数据数组,原表数组,线路,工作地点和内容,工作内容保存,计划编号保存,实际开始时间,实际结束时间,线路名,需要匹配的月份)
                    
                ElseIf bret=True And 计划结束月份>6 And 实际月份>6
                    回填 = 完成状态回填(新数据数组,原表数组,线路,工作地点和内容,工作内容保存,计划编号保存,实际开始时间,实际结束时间,线路名,需要匹配的月份)
                End If
            ElseIf 电压等级 = "220kV" //一季度一测
                If 计划结束月份 = 1 or 计划结束月份 = 3
                    section = 1
                ElseIf 计划结束月份 = 4 or 计划结束月份 = 6
                    section = 2
                ElseIf 计划结束月份 = 7 or 计划结束月份 = 9
                    section = 3
                ElseIf 计划结束月份 = 10 or 计划结束月份 = 12
                    section = 4
                End If

                If bret=True And section = 1 And (实际月份 = 1 Or 实际月份 = 3)
                    回填 = 完成状态回填(新数据数组,原表数组,线路,工作地点和内容,工作内容保存,计划编号保存,实际开始时间,实际结束时间,线路名,需要匹配的月份)
                ElseIf bret=True And section = 2 And (实际月份 = 4 Or 实际月份 = 6)
                    回填 = 完成状态回填(新数据数组,原表数组,线路,工作地点和内容,工作内容保存,计划编号保存,实际开始时间,实际结束时间,线路名,需要匹配的月份)
                ElseIf bret=True And section = 3 And (实际月份 = 7 Or 实际月份 = 9)
                    回填 = 完成状态回填(新数据数组,原表数组,线路,工作地点和内容,工作内容保存,计划编号保存,实际开始时间,实际结束时间,线路名,需要匹配的月份)
                ElseIf bret=True And section = 4 And (实际月份 = 10 Or 实际月份 = 12)
                    回填 = 完成状态回填(新数据数组,原表数组,线路,工作地点和内容,工作内容保存,计划编号保存,实际开始时间,实际结束时间,线路名,需要匹配的月份)
                End If
            End If
        End If
	Next
    Return "已填写关注和一般"
End Function
