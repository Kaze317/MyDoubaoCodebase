Function 最近一次完成(objExcelyuanbiao,i,当前月份,预试计划表子表名) 
    //回填没完成的上一个月的完成时间
    If 当前月份 = 8 
        最近一次完成时间 = Excel.ReadCell(objExcelyuanbiao,预试计划表子表名,"U"&i,true) //七月
    ElseIf 当前月份 = 9
        最近一次完成时间 = Excel.ReadCell(objExcelyuanbiao,预试计划表子表名,"W"&i,true) //八月
    ElseIf 当前月份 = 10
        最近一次完成时间 = Excel.ReadCell(objExcelyuanbiao,预试计划表子表名,"Y"&i,true) //九月
    ElseIf 当前月份 = 11
        最近一次完成时间 = Excel.ReadCell(objExcelyuanbiao,预试计划表子表名,"AA"&i,true) //十月
    ElseIf 当前月份 = 12
        最近一次完成时间 = Excel.ReadCell(objExcelyuanbiao,预试计划表子表名,"AC"&i,true) //十一月
    End If
    Return 最近一次完成时间
End Function

Function 新建工作表(核查数据表,工作表名) 
    Excel.CreateSheet(核查数据表,工作表名,"after",false)
    Excel.ActiveSheet(核查数据表,工作表名)
End Function

Function 各中心统计(中心名称) 
    If Regex.Test(中心名称,"白云") = True 
        白云中心 = 白云中心 + 1
        // 白云 = "白云中心"
    ElseIf Regex.Test(中心名称,"花都") = True 
        花都中心 = 花都中心 + 1
        // 花都 = "花都中心"
    ElseIf Regex.Test(中心名称,"从化") = True 
        从化中心 = 从化中心 + 1
        // 从化 = "从化中心"
    ElseIf Regex.Test(中心名称,"天河") = True 
        天河中心 = 天河中心 + 1
        // 天河 = "天河中心"
    End If
End Function

Function 涉及中心(统计)
    涉及 = ""
    //[白云中心,花都中心,从化中心,天河中心]
    If 统计[0] <> 0
        涉及 = 涉及 + "白云中心"
    End If
    If 统计[1] <> 0
        涉及 = 涉及 + "花都中心"
    End If
    If 统计[2] <> 0
        涉及 = 涉及 + "从化中心"
    End If
    If 统计[3] <> 0
        涉及 = 涉及 + "天河中心"
    End If
    Return 涉及
End Function

Function 替换(内容) 
    内容 = Replace(内容,"白云中心","白云中心、",false)
    内容 = Replace(内容,"花都中心","花都中心、",false)
    内容 = Replace(内容,"从化中心","从化中心、",false)
    内容 = Replace(内容,"天河中心","天河中心、",false)
    Return 内容
End Function

Function 重要交叉跨越测温(objExcelyuanbiao,核查数据表,当前月份) 
    number = 2
    row = Excel.GetRowsCount(objExcelyuanbiao,"重要跨越区段测温")
    预试计划表子表名 = "重要跨越区段测温"
    工作表名 = "1.重要交叉跨越测温"
    新建 = 新建工作表(核查数据表,工作表名) 
    Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","A1","区段编号",false)
    Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","B1","线路名称",false)
    Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","C1","杆塔区段",false)
    Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","D1","所属班组",false)
    Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","E1","最近1次完成时间",false)
    Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","F1","到期提醒",false)
    Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","G1","跨越等级",false)
    读取区域内容 = Excel.ReadRange(objExcelyuanbiao,"重要跨越区段测温","A4:H"&row,true)
    // TracePrint(读取区域内容[0][4])
    For i = 4 To row step 1
        到期提醒 = Excel.ReadCell(objExcelyuanbiao,"重要跨越区段测温","AG"&i,true)
        If 到期提醒 = 当前月份&"月份需要提醒"
            Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","A"&number,读取区域内容[i-4][3],false)
            Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","B"&number,读取区域内容[i-4][1],false)
            Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","C"&number,读取区域内容[i-4][2],false)
            Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","D"&number,读取区域内容[i-4][4],false)
            跨越等级 = Excel.ReadCell(objExcelyuanbiao,"重要跨越区段测温","AH"&i,true)
            Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","F"&number,到期提醒,false)
            Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","G"&number,跨越等级,false)
            //回填时间
            // If 最近一次完成时间 = "已取消" 
            //     Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","E"&number,最近一次完成时间,false)
            // End If
            最近一次完成时间 = 最近一次完成(objExcelyuanbiao,i,当前月份,预试计划表子表名) 
            本月最近一次完成时间 = 本月最近一次完成(objExcelyuanbiao,i,当前月份,预试计划表子表名) 
            If 最近一次完成时间 <> "" 
                Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","E"&number,最近一次完成时间,false)
            Else 
                Excel.WriteCell(核查数据表,"1.重要交叉跨越测温","E"&number,本月最近一次完成时间,false)
            End If
            number = number + 1 
            中心名称 = 读取区域内容[i-4][4]
            中心统计 = 各中心统计(中心名称) 
        End If
    Next
    统计 = [白云中心,花都中心,从化中心,天河中心]
    涉及 = 涉及中心(统计)
    白云中心 = 0
    从化中心 = 0
    花都中心 = 0
    天河中心 = 0
    Return 涉及
End Function
