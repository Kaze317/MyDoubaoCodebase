Function 两月一测(计划结束月份,当前月份)
    If (计划结束月份 = 1 or 计划结束月份 = 2)
        section = 2
    ElseIf (计划结束月份 = 3 or 计划结束月份 = 4)
        section = 4
    ElseIf (计划结束月份 = 5 or 计划结束月份 = 6)
        section = 6
    ElseIf (计划结束月份 = 7 or 计划结束月份 = 8)
        section = 8
    ElseIf (计划结束月份 = 9 or 计划结束月份 = 10)
        section = 10
    ElseIf (计划结束月份 = 11 or 计划结束月份 = 12)
        section = 12
    End If

    If 当前月份 = 1 Or 当前月份 = 3 Or 当前月份 = 5 Or 当前月份 = 7 Or 当前月份 = 9 Or 当前月份 = 11
        两月一测当前月份 =  当前月份 + 1
    else
        两月一测当前月份 = 当前月份
    End If
    两月 = [section,两月一测当前月份]
    Return 两月
    
End Function

End Function

Function 其余时段(计划结束月份,当前月份,其余时段匹配,完成状态)

        Return "其余时段成立"
    End If
End Function

Function 电缆护套环流检测(objExcelyuanbiao,当前日期,当前年份,当前月份) 
    //读取线路名和结束时间 并筛选出月份，最后转成数组
    原表 = Excel.ReadRange(objExcelyuanbiao, "5、电缆护套环流检测", "D4:H815", false)
    原表 = Datatable.BuildDataTable(原表, ["线路名称", "线路重要度", "工作内容", "计划开始日期", "计划结束日期"])
    //创建新的数据表
    原表 = Datatable.SelectDataTableColumns(原表, ["线路名称", "计划开始日期", "计划结束日期","线路重要度"])
    arrayShape = Datatable.GetDataTableShape(原表)
    原表数组 = Datatable.GetDataTableByArray(原表, false)

    For i = 0 To UBound(原表数组) step 1
        线路重要度 = 原表数组[i][3]
        其余时段匹配 = Regex.Test(线路重要度,"其余时段")
        完成状态 = Excel.ReadCell(objExcelyuanbiao,"5、电缆护套环流检测","X"&i+4,true)
        计划结束年份 = Time.Year(Time.CDate(原表数组[i][2], "yyyy.mm.dd"))
        计划结束月份 = Time.Month(Time.CDate(原表数组[i][2], "yyyy.mm.dd"))

        两月 = 两月一测(计划结束月份,当前月份)
        六月 = 六月一测(计划结束月份,当前月份)
        其余 = 其余时段(计划结束月份,当前月份,其余时段匹配,完成状态) 
        If 计划结束年份 =  当前年份
            If 两月[0] = 两月[1] And 完成状态 = "" And (线路重要度 = "关键" Or 线路重要度 = "重要") And 两月[0] = 当前月份
                Excel.WriteCell(objExcelyuanbiao,"5、电缆护套环流检测","Z"&i+4,当前月份&"月份需要提醒",false)
            ElseIf 其余 = "其余时段成立" 
                Excel.WriteCell(objExcelyuanbiao,"5、电缆护套环流检测","Z"&i+4,计划结束月份&"月份需要提醒",false)
            ElseIf 六月[0] = 当前月份 And 完成状态 = "" And (线路重要度 = "一般" Or 线路重要度 = "关注") 
                Excel.WriteCell(objExcelyuanbiao,"5、电缆护套环流检测","Z"&i+4,当前月份&"月份需要提醒",false)
            End If
        End If
    Next
End Function
